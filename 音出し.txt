<script>
  // GitHub raw URL を貼る（ここを差し替えるだけ）
  const soundURLs = {
    1: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/junan.mp3",
    2: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/plank.mp3",
    3: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/gai.mp3",
    4: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/furi.mp3",
    5: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/spin30.mp3",
    6: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/spin60.mp3",
    7: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/ashidashi.mp3",
    8: "https://raw.githubusercontent.com/ユーザー名/リポジトリ名/main/noriwaza.mp3"
  };

  // Audio を事前に読み込む（遅延防止）
  const audios = {};
  for (let i = 1; i <= 8; i++) {
    audios[i] = new Audio(soundURLs[i]);
  }

  // ボタンにイベントを付与
  document.querySelectorAll("button").forEach(btn => {
    btn.addEventListener("click", () => {
      const id = btn.dataset.sound;
      const audio = audios[id];
      audio.currentTime = 0; // 連打対応
      audio.play();
    });
  });
</script>
